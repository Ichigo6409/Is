<?php

namespace App\Http\Controllers;

use App\Http\Requests\StoreProductRequest;
use App\Http\Requests\UpdateProductRequest;
use App\Models\Product;
use App\Models\Category;
use Inertia\Inertia;
use Inertia\Response;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;

class ProductController extends Controller
{
    protected string $businessId = 'BUS-CD-SOUV-001';

    public function index(Request $request): Response
    {
        // only_category es opcional: si viene (ej. desde otra vista que
        // reutilice este mismo listado), filtra por esa categoría. Si no
        // viene, se listan todos los productos como antes.
        $onlyCategory = trim((string) $request->query('only_category', ''));
        $onlyCategory = $onlyCategory !== '' ? $onlyCategory : null;

        return $this->buildProductsIndex($request, $onlyCategory, 'Equipo4/Productos');
    }

    public function souvenirs(Request $request): Response
    {
        // Souvenirs NO tiene CRUD propio: es Productos filtrado a
        // category = 'souvenirs', reutilizando el mismo query/paginación/
        // búsqueda que index().
        return $this->buildProductsIndex($request, 'souvenirs', 'Equipo4/Souvenirs');
    }

    /**
     * Construye el listado paginado de productos, opcionalmente filtrado
     * por categoría, y lo renderiza en la página Inertia indicada.
     * Centraliza la lógica que comparten index() y souvenirs() para no
     * duplicar el query, el mapeo de datos ni la paginación.
     */
    private function buildProductsIndex(Request $request, ?string $onlyCategory, string $page): Response
    {
        $q = trim((string) $request->query('q', ''));
        $perPage = 25;

        $query = Product::where('business_id', $this->businessId);

        if ($onlyCategory !== null) {
            $query->where('category', $onlyCategory);
        }

        if ($q !== '') {
            $regex = '/' . preg_quote($q, '/') . '/i';
            $query->where(function ($sub) use ($regex) {
                $sub->where('sku', 'regex', $regex)
                    ->orWhere('name', 'regex', $regex);
            });
        }

        $products = $query->orderBy('created_at', 'desc')->paginate($perPage);

        $productsData = collect($products->items())->map(function ($product) {
            return [
                '_id' => (string) $product->_id,
                'sku' => (string) $product->sku,
                'name' => (string) $product->name,
                'description' => (string) ($product->description ?? ''),
                'category' => (string) $product->category,
                'stock_min' => (int) $product->stock_min,
                'stock_max' => (int) $product->stock_max,
                'active' => (bool) $product->active,
            ];
        })->values()->all();

        $categories = Category::where('business_id', $this->businessId)
            ->where('active', true)
            ->orderBy('name', 'asc')
            ->get()
            ->map(fn($cat) => [
                'slug' => (string) $cat->slug,
                'name' => (string) $cat->name,
            ])
            ->values()
            ->all();

        return Inertia::render($page, [
            'products' => $productsData,
            'categories' => $categories,
            'pagination' => [
                'current_page' => $products->currentPage(),
                'last_page' => $products->lastPage(),
                'per_page' => $products->perPage(),
                'total' => $products->total(),
                'from' => $products->firstItem() ?? 0,
                'to' => $products->lastItem() ?? 0,
            ],
            'filters' => ['q' => $q],
        ]);
    }

    public function store(StoreProductRequest $request): RedirectResponse
    {
        $validated = $request->validated();
        $validated['business_id'] = $this->businessId;

        Product::create($validated);

        return redirect()->route('equipo4.productos.index')->with('success', 'Producto creado exitosamente.');
    }

    public function update(UpdateProductRequest $request, string $id): RedirectResponse
    {
        $product = Product::where('_id', $id)
            ->where('business_id', $this->businessId)
            ->first();

        if (!$product) {
            abort(404, 'Producto no encontrado.');
        }

        $product->update($request->validated());

        return redirect()->route('equipo4.productos.index')->with('success', 'Producto actualizado exitosamente.');
    }

    public function destroy(string $id): RedirectResponse
    {
        $product = Product::where('_id', $id)
            ->where('business_id', $this->businessId)
            ->first();

        if (!$product) {
            abort(404, 'Producto no encontrado.');
        }

        $product->delete();

        return redirect()->route('equipo4.productos.index')->with('success', 'Producto eliminado exitosamente.');
    }
}

{
    protected string $businessId = 'BUS-CD-SOUV-001';

    public function index(Request $request): Response
    {
        $q = trim((string) $request->query('q', ''));
        $type = trim((string) $request->query('type', ''));
        $perPage = 25;

        $query = StockMovement::where('business_id', $this->businessId);

        if ($q !== '') {
            $regex = '/' . preg_quote($q, '/') . '/i';
            $query->where(function ($sub) use ($regex) {
                $sub->where('external_reference', 'regex', $regex)
                    ->orWhere('reason', 'regex', $regex);
            });

        if ($type !== '') {
            $query->where('type', $type);

                'type' => $type,
            ],
        ]);
    }

namespace App\Http\Controllers;

use App\Models\Product;

use Illuminate\Support\Facades\Log;
use Inertia\Inertia;
use Inertia\Response;

class ReservationController extends Controller
{
    protected string $businessId = 'BUS-CD-SOUV-001';

    public function index(Request $request): Response
    {
        $q = trim((string) $request->query('q', ''));
        $perPage = 25;

        $query = StockReservation::where('business_id', $this->businessId);

        if ($q !== '') {
            $regex = '/' . preg_quote($q, '/') . '/i';
            $query->where('external_reference', 'regex', $regex);
        }

        $reservations = $query->orderBy('created_at', 'desc')->paginate($perPage);

        $productsMap = Product::where('business_id', $this->businessId)
            ->get()
            ->keyBy(fn($p) => (string) $p->_id);

        $data = collect($reservations->items())->map(function ($r) use ($productsMap) {
            $product = $productsMap->get((string) $r->product_id);
            return [
                '_id' => (string) $r->_id,
                'external_reference' => (string) ($r->external_reference ?? ''),
                'product_sku' => $product ? (string) $product->sku : '',
                'product_name' => $product ? (string) $product->name : '',
                'quantity' => (int) $r->quantity,
                'source' => (string) ($r->source ?? ''),
                'expires_at' => $r->expires_at instanceof \DateTimeInterface
                    ? $r->expires_at->format('d/m/Y H:i')
                    : (string) ($r->expires_at ?? ''),
                'status' => (string) $r->status,
            ];
        })->values()->all();

        return Inertia::render('Equipo4/Reservas', [
            'reservations' => $data,
            'pagination' => [
                'current_page' => $reservations->currentPage(),
                'last_page' => $reservations->lastPage(),
                'per_page' => $reservations->perPage(),
                'total' => $reservations->total(),
                'from' => $reservations->firstItem() ?? 0,
                'to' => $reservations->lastItem() ?? 0,
            ],
            'filters' => [
                'q' => $q,
            ],
        ]);
    }

    public function release(string $id): RedirectResponse
    {
        $reservation = StockReservation::where('_id', $id)
            ->where('business_id', $this->businessId)
            ->first();

        if (!$reservation) {
            return redirect()->route('equipo4.reservas.index')->withErrors(['error' => 'Reserva no encontrada.']);
        }

        if ($reservation->status !== 'RESERVED') {
            return redirect()->route('equipo4.reservas.index')->withErrors(['error' => 'Solo se pueden liberar reservas en estado RESERVED.']);
        }

        $quantity = (int) $reservation->quantity;

        // Sin transacciones porque MongoDB está en standalone: primero se
        // libera el inventario (reserved -= qty, available += qty) y solo si
        // eso funciona se marca la reserva como RELEASED. Si el guardado de
        // la reserva falla después, se compensa revirtiendo el inventario.
        $inventoryCollection = DB::connection('mongodb')->getCollection('inventories');

        $inventoryFilter = [
            'business_id' => $this->businessId,
            'product_id' => (string) $reservation->product_id,
            'variant_id' => $reservation->variant_id ?? null,
            'location_id' => (string) $reservation->location_id,
            '$expr' => [
                '$gte' => ['$reserved', $quantity],
            ],
        ];

        $updatedInventory = $inventoryCollection->findOneAndUpdate(
            $inventoryFilter,
            [
                '$inc' => [
                    'reserved' => -$quantity,
                    'available' => $quantity,
                ],
                '$set' => [
                    'updated_at' => now()->toDateTime(),
                ],
            ]
        );

        if ($updatedInventory === null) {
            return redirect()->route('equipo4.reservas.index')->withErrors(['error' => 'No fue posible liberar el inventario reservado.']);
        }

        try {
            $reservation->status = 'RELEASED';
            $reservation->save();
        } catch (\Throwable $e) {
            Log::error('Fallo al marcar reserva como liberada. Revirtiendo inventario.', [
                'reservation_id' => $id,
                'error' => $e->getMessage(),
            ]);

            try {
                $inventoryCollection->findOneAndUpdate(
                    [
                        'business_id' => $this->businessId,
                        'product_id' => (string) $reservation->product_id,
                        'variant_id' => $reservation->variant_id ?? null,
                        'location_id' => (string) $reservation->location_id,
                    ],
                    [
                        '$inc' => [
                            'reserved' => $quantity,
                            'available' => -$quantity,
                        ],
                        '$set' => [
                            'updated_at' => now()->toDateTime(),
                        ],
                    ]
                );
            } catch (\Throwable $ex) {
                Log::critical('Fallo al revertir inventario tras error al liberar reserva.', [
                    'reservation_id' => $id,
                    'cleanup_error' => $ex->getMessage(),
                ]);
            }

            return redirect()->route('equipo4.reservas.index')->withErrors(['error' => 'No se pudo liberar la reserva.']);
        }

        return redirect()->route('equipo4.reservas.index')->with('success', 'Reserva liberada exitosamente.');
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Product;
use App\Models\ReorderRule;
use App\Models\StockAlert;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Log;
use Inertia\Inertia;
use Inertia\Response;

class AlertController extends Controller
{
    protected string $businessId = 'BUS-CD-SOUV-001';

    public function index(Request $request): Response
    {
        $q = trim((string) $request->query('q', ''));
        $perPage = 25;

        $query = StockAlert::where('business_id', $this->businessId)
            ->where('status', 'ACTIVE');

        $productsMap = Product::where('business_id', $this->businessId)
            ->get()
            ->keyBy(fn($p) => (string) $p->_id);

        if ($q !== '') {
            $regex = '/' . preg_quote($q, '/') . '/i';
            $matchingProductIds = $productsMap->filter(function ($p) use ($regex) {
                return preg_match($regex, (string) $p->sku) || preg_match($regex, (string) $p->name);
            })->map(fn($p) => (string) $p->_id)->values()->all();

            $query->whereIn('product_id', $matchingProductIds);
        }

        $alerts = $query->orderBy('created_at', 'desc')->paginate($perPage);

        // Mínimo y punto de reorden viven en reorder_rules (para poder
        // editar la regla sin tocar el historial de alertas ya generadas).
        // Si una alerta no tiene regla asociada, se usa 'threshold' de la
        // propia alerta como respaldo.
        $rulesMap = ReorderRule::where('business_id', $this->businessId)
            ->get()
            ->keyBy(fn($r) => (string) $r->product_id . '|' . (string) $r->location_id);

        $data = collect($alerts->items())->map(function ($a) use ($productsMap, $rulesMap) {
            $product = $productsMap->get((string) $a->product_id);
            $rule = $rulesMap->get((string) $a->product_id . '|' . (string) $a->location_id);

            return [
                '_id' => (string) $a->_id,
                'product_sku' => $product ? (string) $product->sku : '',
                'product_name' => $product ? (string) $product->name : '',
                'current_qty' => (int) $a->current_qty,
                'min_qty' => $rule ? (int) $rule->min_qty : (int) $a->threshold,
                'reorder_point' => $rule ? (int) $rule->reorder_point : (int) $a->threshold,
                'priority' => (string) $a->priority,
                'status' => (string) $a->status,
            ];
        })->values()->all();

        return Inertia::render('Equipo4/Alertas', [
            'alerts' => $data,
            'pagination' => [
                'current_page' => $alerts->currentPage(),
                'last_page' => $alerts->lastPage(),
                'per_page' => $alerts->perPage(),
                'total' => $alerts->total(),
                'from' => $alerts->firstItem() ?? 0,
                'to' => $alerts->lastItem() ?? 0,
            ],
            'filters' => [
                'q' => $q,
            ],
        ]);
    }

    public function resolve(string $id): RedirectResponse
    {
        $alert = StockAlert::where('_id', $id)
            ->where('business_id', $this->businessId)
            ->first();

        if (!$alert) {
            return redirect()->route('equipo4.alertas.index')->withErrors(['error' => 'Alerta no encontrada.']);
        }

        if ($alert->status !== 'ACTIVE') {
            return redirect()->route('equipo4.alertas.index')->withErrors(['error' => 'Solo se pueden resolver alertas activas.']);
        }

        try {
            $alert->status = 'RESOLVED';
            $alert->save();
        } catch (\Throwable $e) {
            Log::error('Fallo al marcar alerta como resuelta.', [
                'alert_id' => $id,
                'error' => $e->getMessage(),
            ]);
            return redirect()->route('equipo4.alertas.index')->withErrors(['error' => 'No se pudo marcar la alerta como resuelta.']);
        }

        return redirect()->route('equipo4.alertas.index')->with('success', 'Alerta marcada como resuelta.');
    }
}
